public class IntegerExample {

    public static void main(String[] args) {

        // 1. Conversion: String to Primitive
        String ageStr = "25";
        int age = Integer.parseInt(ageStr);

        System.out.println("String to Integer: " + age);


        // 2. Conversion: Primitive to String
        String ageBack = Integer.toString(age);

        System.out.println("Integer to String: " + ageBack);


        // 3. Constant Values
        System.out.println("Max Integer: " + Integer.MAX_VALUE);
        System.out.println("Min Integer: " + Integer.MIN_VALUE);


        // 4. Base Conversion
        System.out.println("Binary of 10: " + Integer.toBinaryString(10));
        System.out.println("Hex of 255: " + Integer.toHexString(255));


        // 5. Comparison
        Integer a = 10;
        Integer b = 20;

        System.out.println("Comparison: " + Integer.compare(a, b));
    }
}

import java.util.HashMap;
import java.util.Map;
import java.util.Iterator;
import java.util.Set;

public class HashMapDemo {

    public static void main(String args[]) {

        HashMap<Integer, String> hmap = new HashMap<Integer, String>();

        // Adding elements to HashMap
        hmap.put(12, "Bishnu");
        hmap.put(2, "Santosh");
        hmap.put(7, "Prakash");
        hmap.put(49, "Bhupi");
        hmap.put(3, "Mahendra");

        // Displaying key and value using Iterator
        Set<Map.Entry<Integer, String>> set = hmap.entrySet();

        Iterator<Map.Entry<Integer, String>> iterator = set.iterator();

        while (iterator.hasNext()) {
            Map.Entry<Integer, String> mentry = iterator.next();

            System.out.print("Key is: " + mentry.getKey());
            System.out.println(" & Value is: " + mentry.getValue());
        }

        // Getting value using key
        String var = hmap.get(2);

        System.out.println("Value at key 2: " + var);

        // Removing an element
        hmap.remove(3);

        System.out.println("Map key and values after removal:");

        Set<Map.Entry<Integer, String>> set2 = hmap.entrySet();

        Iterator<Map.Entry<Integer, String>> iterator2 = set2.iterator();

        while (iterator2.hasNext()) {

            Map.Entry<Integer, String> mentry2 = iterator2.next();

            System.out.print("Key is: " + mentry2.getKey());
            System.out.println(" & Value is: " + mentry2.getValue());
        }
    }
}

import java.util.*;

public class ArrayListDemo {

    public static void main(String[] args) {

        List<Integer> l1 = new ArrayList<Integer>();

        l1.add(0, 1); // Adds 1 at index 0
        l1.add(1, 2); // Adds 2 at index 1

        System.out.println(l1); // [1, 2]

        List<Integer> l2 = new ArrayList<Integer>();

        l2.add(1);
        l2.add(2);
        l2.add(3);

        l1.addAll(1, l2);

        System.out.println(l1); // [1, 1, 2, 3, 2]

        l1.remove(1);

        System.out.println(l1); // [1, 2, 3, 2]

        System.out.println(l1.get(3)); // 2

        l1.set(0, 5);

        System.out.println(l1); // [5, 2, 3, 2]
    }
}

import java.util.ArrayList;
import java.util.ListIterator;

public class ListIteratorDemo {

    public static void main(String[] args) {

        ArrayList<Integer> al = new ArrayList<Integer>();

        // Adding numbers from 0 to 9
        for (int i = 0; i < 10; i++) {
            al.add(i);
        }

        System.out.println(al);

        // Creating ListIterator
        ListIterator<Integer> itr = al.listIterator();

        while (itr.hasNext()) {

            int i = itr.next();

            System.out.print(i + " ");

            if (i % 2 == 0) {

                i++;          // Change even number to odd
                itr.set(i);   // Change current value
                itr.add(i);   // Add the new value
            }
        }

        System.out.println();

        System.out.println(al);
    }
}

    }

import java.util.*;

public class IteratorDemo {

    public static void main(String[] args) {

        List<String> names = new ArrayList<>(
            Arrays.asList("Apple", "Banana", "Mango", "Orange")
        );

        // Get the iterator
        Iterator<String> it = names.iterator();

        while (it.hasNext()) {

            String name = it.next();

            if (name.equals("Banana")) {
                it.remove(); // Safely removes "Banana"
            } else {
                System.out.println(name);
            }
        }

        System.out.println("List after removal: " + names);

# The comparator Interface 

import java.util.*;

class Employee {

    String name;
    int salary;

    Employee(String name, int salary) {
        this.name = name;
        this.salary = salary;
    }
}

class SalaryComparator implements Comparator<Employee> {

    public int compare(Employee e1, Employee e2) {
        return e1.salary - e2.salary;
    }
}

public class ComparatorDemo {

    public static void main(String[] args) {

        List<Employee> list = new ArrayList<>();

        list.add(new Employee("John", 5000));
        list.add(new Employee("Alice", 7000));

        // Sort using our custom Comparator
        Collections.sort(list, new SalaryComparator());

        for (Employee e : list) {
            System.out.println(e.name + " : " + e.salary);
        }
    }
}

class Box<T> {

    private T content;

    public void setContent(T content) {
        this.content = content;
    }

    public T getContent() {
        return content;
    }
}

public class Main {

    public static void main(String[] args) {

        // Create a Box for Integers
        Box<Integer> intBox = new Box<>();
        intBox.setContent(100);

        // Create a Box for Strings
        Box<String> strBox = new Box<>();
        strBox.setContent("Hello Generics");

        // Display values
        System.out.println("Integer Value: " + intBox.getContent());
        System.out.println("String Value: " + strBox.getContent());
    }
}
