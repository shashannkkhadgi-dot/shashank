public class factorial {
    public static void main(String[] agre) {
        Scanner sc = new Scanner(System.in);

        System.out.println("Enter the number:");
        int n = sc.nextInt();

        int factorial = 1;

        if (n >= 0) {
            for (int i = 1; i <= n; i++) {
